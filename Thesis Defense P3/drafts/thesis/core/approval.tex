% The university's approval page is one page carrying the whole examining committee.
% With the role label stacked above its signature rule, the four blocks need about
% 23 cm and the text area is 24.7 cm less the heading, so the fourth signature was
% pushed onto a page of its own. \sigblock puts the role at the left and the rule at
% the right of the same band, which is how the university's own page is laid out and
% gives back two lines per member. The signing gap stays close to the declaration's.
\setlength\sigskip{1.25cm}

% The optional argument is a cropped signature file. It goes through \esignhere
% from the declaration page, which gives it no width and no height, so it sits in
% the blank signing gap that is already there and every block keeps exactly the
% size it had unsigned. That matters here more than anywhere: this page only just
% holds four signatures, and any real height would push the fourth off it again.
\newcommand*\sigblock[3][]{%
  \par\vspace{0.45cm}%
  \noindent
  \begin{minipage}[t]{4.2cm}\setlength\parskip{0pt}#2\end{minipage}%
  \hfill
  \begin{minipage}[t]{10.5cm}\setlength\parskip{0pt}%
    \vspace*{\sigskip}%
    \ifx\relax#1\relax\else\esignhere{#1}\fi
    \hrulefill\par#3%
  \end{minipage}%
  \par}
