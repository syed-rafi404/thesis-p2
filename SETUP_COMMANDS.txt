# Multimodal Banglish Classroom Summarizer
# Windows 11 | RTX 3060 (12GB) / RTX 3090 (24GB) | CUDA 12.4
# =============================================================================

# -----------------------------------------------------------------------------
# STEP 1: Create Fresh Conda Environment
# -----------------------------------------------------------------------------
conda create -n thesis_v2 python=3.10 -y
conda activate thesis_v2

# -----------------------------------------------------------------------------
# STEP 2: Install PyTorch with CUDA 12.4
# -----------------------------------------------------------------------------
pip install torch torchvision torchaudio --index-url https://download.pytorch.org/whl/cu124

# -----------------------------------------------------------------------------
# STEP 3: Install All Dependencies from requirements.txt
# -----------------------------------------------------------------------------
pip install -r requirements.txt

# -----------------------------------------------------------------------------
# STEP 4: Install bitsandbytes for 4-bit quantization (REQUIRED for 12GB GPU)
# -----------------------------------------------------------------------------
# Note: bitsandbytes is REQUIRED for 4-bit quantization on RTX 3060 (12GB)
pip install bitsandbytes>=0.42.0

# -----------------------------------------------------------------------------
# STEP 5: Verify Installation
# -----------------------------------------------------------------------------
python verify_stack.py

# -----------------------------------------------------------------------------
# ALTERNATIVE: Manual Installation (if requirements.txt fails)
# -----------------------------------------------------------------------------

# Transformers & Acceleration Stack
pip install transformers accelerate optimum safetensors sentencepiece

# Qwen2.5-VL Dependencies (VLM)
pip install qwen-vl-utils

# Audio Processing
pip install librosa soundfile pydub

# Vision Processing
pip install opencv-python pillow

# Video Processing
pip install moviepy

# Pose Estimation (Gaze Tracking)
pip install ultralytics

# Evaluation
pip install thefuzz jiwer

# Utilities
pip install tqdm pyyaml python-dotenv rich pytest

# =============================================================================
# QUICK TEST - Run the Full Pipeline
# =============================================================================

# Process a single video (all 3 novelties) - 12GB GPU uses 4-bit by default
python run_thesis.py "data/raw/lecture.mp4" -o "output/test" --interval 30 --live

# Live mode with 4-bit VLM and LLM (fits all models in 12GB VRAM)
python run_thesis.py "data/raw/lecture.mp4" --live --interval 45

# Batch process all videos in data/raw/
python batch_process.py --live --interval 45

# GPU MEMORY REQUIREMENTS

#
# RTX 3060 (12GB VRAM) - Config: config/live_config.yaml
#   Whisper large-v3-turbo:  ~3.0 GB (FP16)
#   Qwen2.5-VL-7B 4-bit:     ~4.5 GB (quantized)
#   Qwen2.5-7B 4-bit:        ~4.0 GB (quantized)
#   YOLOv8-Pose:             ~0.5 GB
#   ─────────────────────────────────────
#   Total:                   ~12.0 GB ✅
#
# RTX 3090/4090 (24GB VRAM) - Can use FP16 VLM for better OCR:
#   Set vlm.precision: "fp16" in config/live_config.yaml
#
# =============================================================================
# TROUBLESHOOTING
# =============================================================================
# 
# Issue: "CUDA out of memory" on RTX 3060
# Fix: 
#   1. Ensure --live mode is used (enables 4-bit quantization)
#   2. Close other GPU applications (browser, other models)
#   3. Reduce frame_interval to process fewer frames (e.g., --interval 60)
#
# Issue: "bitsandbytes not working on Windows"
# Fix: pip install bitsandbytes>=0.42.0 (recent versions work on Windows)
#      This is REQUIRED for 4-bit quantization on 12GB GPUs
#
# Issue: "torch.cuda.is_available() returns False"
# Fix: 
#   1. Verify NVIDIA drivers are installed: nvidia-smi
#   2. Reinstall PyTorch with correct CUDA version
#   3. Check CUDA_PATH environment variable
#
# Issue: "No module named 'transformers'"
# Fix: Ensure conda environment is activated: conda activate thesis_v2
#
# Issue: "ultralytics / YOLOv8 not found"
# Fix: pip install ultralytics
#
